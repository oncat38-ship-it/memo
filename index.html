        function getMimeTypeByExt(ext) {

                'csv': 'text/csv;charset=utf-8',
                'md': 'text/markdown;charset=utf-8',

                'xml': 'application/xml;charset=utf-8',

        function isFinderEye(r, c, count) {
            if (r < 7 && c < 7) return true;
            if (r < 7 && c >= count - 7) return true;
            if (r >= count - 7 && c < 7) return true;

        function generateCode() {
            const text = document.getElementById('code-text').value.trim() || 'https://example.com';
            document.getElementById('code-text-length').innerText = `${text.length} 文字`;

            const canvas = document.getElementById('code-canvas');
            const svg = document.getElementById('barcode-svg');

            if (currentCodeType === 'qr') {
                canvas.classList.remove('hidden');
                svg.classList.add('hidden');

                const dotStyle = document.getElementById('qr-dot-style').value;
                const fgColor = document.getElementById('qr-fg-color').value;
                const gradColor = document.getElementById('qr-grad-color').value;
                const eyeColor = document.getElementById('qr-eye-color').value;
                const bgColor = document.getElementById('qr-bg-color').value;
                const useGradient = document.getElementById('qr-use-gradient').checked;

                try {
                    const qr = qrcode(0, 'H');
                    qr.addData(text);
                    qr.make();

                    const count = qr.getModuleCount();
                    const cellSize = 8;
                    const margin = 20;
                    const size = count * cellSize + margin * 2;

                    canvas.width = size;
                    canvas.height = size;

                    const ctx = canvas.getContext('2d');
                    ctx.fillStyle = bgColor;
                    ctx.fillRect(0, 0, size, size);

                    let bodyFillStyle = fgColor;
                    if (useGradient) {
                        const grad = ctx.createLinearGradient(0, 0, size, size);
                        grad.addColorStop(0, fgColor);
                        grad.addColorStop(1, gradColor);
                        bodyFillStyle = grad;
                    }

                    for (let r = 0; r < count; r++) {
                        for (let c = 0; c < count; c++) {
                            if (qr.isDark(r, c) && !isFinderEye(r, c, count)) {
                                const x = c * cellSize + margin;
                                const y = r * cellSize + margin;
                                ctx.fillStyle = bodyFillStyle;
                                drawCustomDot(ctx, x, y, cellSize, dotStyle);
                            }
                        }
                    }

                    drawEyePattern(ctx, margin, margin, cellSize, eyeColor);
                    drawEyePattern(ctx, (count - 7) * cellSize + margin, margin, cellSize, eyeColor);
                    drawEyePattern(ctx, margin, (count - 7) * cellSize + margin, cellSize, eyeColor);

                    if (overlayLogoImage) {
                        const logoSize = size * 0.22;
                        const logoX = (size - logoSize) / 2;
                        const logoY = (size - logoSize) / 2;

                        ctx.fillStyle = bgColor;
                        ctx.beginPath();
                        ctx.roundRect(logoX - 4, logoY - 4, logoSize + 8, logoSize + 8, 8);
                        ctx.fill();

                        ctx.drawImage(overlayLogoImage, logoX, logoY, logoSize, logoSize);
                    }
                } catch (e) {}
            } else {
                canvas.classList.add('hidden');
                svg.classList.remove('hidden');
                try {
                    JsBarcode("#barcode-svg", text, {
                        format: document.getElementById('barcode-format').value,
                        lineColor: "#000",
                        width: 1.8,
                        height: 50,
                        displayValue: true
                    });
                } catch (e) {}
            }
        }

        function drawCustomDot(ctx, x, y, s, style) {
            ctx.beginPath();
            if (style === 'dots') {
                ctx.arc(x + s / 2, y + s / 2, s / 2.2, 0, Math.PI * 2);
                ctx.fill();
            } else if (style === 'rounded') {
                ctx.roundRect(x, y, s, s, s * 0.3);
                ctx.fill();
            } else if (style === 'wave') {
                ctx.arc(x + s / 2, y + s / 2, s / 2, 0, Math.PI);
                ctx.fill();
            } else if (style === 'liquid') {
                ctx.roundRect(x, y, s, s, [s * 0.5, 0, s * 0.5, 0]);
                ctx.fill();
            } else if (style === 'line') {
                ctx.fillRect(x + s * 0.2, y, s * 0.6, s);
            } else if (style === 'cross') {
                ctx.fillRect(x + s * 0.3, y, s * 0.4, s);
                ctx.fillRect(x, y + s * 0.3, s, s * 0.4);
            } else if (style === 'hexagon') {
                ctx.moveTo(x + s / 2, y);
                ctx.lineTo(x + s, y + s / 4);
                ctx.lineTo(x + s, y + s * 3 / 4);
                ctx.lineTo(x + s / 2, y + s);
                ctx.lineTo(x, y + s * 3 / 4);
                ctx.lineTo(x, y + s / 4);
                ctx.closePath();
                ctx.fill();
            } else if (style === 'diamond') {
                ctx.moveTo(x + s / 2, y);
                ctx.lineTo(x + s, y + s / 2);
                ctx.lineTo(x + s / 2, y + s);
                ctx.lineTo(x, y + s / 2);
                ctx.closePath();
                ctx.fill();
            } else if (style === 'star') {
                drawStar(ctx, x + s/2, y + s/2, 5, s/2, s/4);
            } else {
                ctx.fillRect(x, y, s, s);
            }
        }

        function drawEyePattern(ctx, x, y, cellSize, color) {
            ctx.fillStyle = color;
            ctx.beginPath();
            ctx.roundRect(x, y, 7 * cellSize, 7 * cellSize, cellSize * 1.5);
            ctx.fill();

            ctx.fillStyle = document.getElementById('qr-bg-color').value;
            ctx.beginPath();
            ctx.roundRect(x + cellSize, y + cellSize, 5 * cellSize, 5 * cellSize, cellSize);
            ctx.fill();

            ctx.fillStyle = color;
            ctx.beginPath();
            ctx.roundRect(x + 2 * cellSize, y + 2 * cellSize, 3 * cellSize, 3 * cellSize, cellSize * 0.8);
            ctx.fill();
        }

        function drawStar(ctx, cx, cy, spikes, outerRadius, innerRadius) {
            let rot = Math.PI / 2 * 3;
            let x = cx, y = cy, step = Math.PI / spikes;
            ctx.beginPath();
            ctx.moveTo(cx, cy - outerRadius);
            for (let i = 0; i < spikes; i++) {
                x = cx + Math.cos(rot) * outerRadius;
                y = cy + Math.sin(rot) * outerRadius;
                ctx.lineTo(x, y);
                rot += step;
                x = cx + Math.cos(rot) * innerRadius;
                y = cy + Math.sin(rot) * innerRadius;
                ctx.lineTo(x, y);
                rot += step;
            }
            ctx.lineTo(cx, cy - outerRadius);
            ctx.closePath();
            ctx.fill();
        }

            closeSaveModal();
            alert('クラウドストレージに保存しました！');
        }

        function setStorageGroup(group) {
            currentStorageGroup = group;
            document.querySelectorAll('.group-filter-btn').forEach(b => {
                b.classList.remove('bg-indigo-600', 'text-white');
                b.classList.add('bg-slate-100', 'dark:bg-slate-800', 'text-slate-600', 'dark:text-slate-300');
            });
            document.getElementById(`group-btn-${group}`).classList.remove('bg-slate-100', 'dark:bg-slate-800', 'text-slate-600', 'dark:text-slate-300');
            document.getElementById(`group-btn-${group}`).classList.add('bg-indigo-600', 'text-white');
            renderSavedAssets();
        }

        function filterByTag(tag) {
            currentSelectedTag = currentSelectedTag === tag ? null : tag;
            renderSavedAssets();
        }

        function renderSavedAssets() {
            const container = document.getElementById('saved-assets-list');
            const tagsContainer = document.getElementById('storage-tags-filter');
            const countEl = document.getElementById('saved-assets-count');
            const searchQuery = document.getElementById('storage-search-input').value.toLowerCase();

            container.innerHTML = '';
            tagsContainer.innerHTML = '';

            const allTags = [...new Set(window.savedAssets.flatMap(item => item.tags || []))];
            
            if (allTags.length > 0) {
                allTags.forEach(tag => {
                    const active = currentSelectedTag === tag;
                    const chip = document.createElement('button');
                    chip.className = `px-2 py-0.5 rounded-md text-[10px] font-medium transition ${active ? 'bg-indigo-600 text-white' : 'bg-slate-100 dark:bg-slate-800 text-slate-500 hover:bg-slate-200'}`;
                    chip.innerText = `#${tag}`;
                    chip.onclick = () => filterByTag(tag);
                    tagsContainer.appendChild(chip);
                });
            } else {
                tagsContainer.innerHTML = `<span class="text-[10px] text-slate-400">タグ未設定</span>`;
            }

            const filtered = window.savedAssets.filter(item => {
                const matchesGroup = currentStorageGroup === 'all' || item.type === currentStorageGroup;
                const matchesTag = !currentSelectedTag || (item.tags && item.tags.includes(currentSelectedTag));
                const matchesSearch = !searchQuery || item.title.toLowerCase().includes(searchQuery) || (item.content && item.content.toLowerCase().includes(searchQuery));
                return matchesGroup && matchesTag && matchesSearch;
            });

            countEl.innerText = `${filtered.length} 件`;

            if (filtered.length === 0) {
                container.innerHTML = `<div class="text-center py-8 text-slate-400 text-xs">該当する保存データはありません</div>`;
                return;
            }

            filtered.forEach((item, index) => {
                const card = document.createElement('div');
                card.className = "bg-white dark:bg-slate-900 rounded-xl border border-slate-200 dark:border-slate-800 p-2.5 shadow-sm space-y-2";

                const tagsHtml = (item.tags || []).map(t => `<span class="bg-slate-100 dark:bg-slate-800 text-[9px] px-1.5 py-0.5 rounded text-indigo-500 font-medium">#${escapeHtml(t)}</span>`).join(' ');

                if (item.type === 'media') {
                    card.innerHTML = `
                        <div class="flex items-center justify-between">
                            <div class="flex items-center space-x-2.5">
                                <img src="${item.data}" class="w-10 h-10 object-contain rounded bg-slate-50 dark:bg-slate-950 border border-slate-200 dark:border-slate-800">
                                <div>
                                    <h4 class="font-bold text-xs max-w-[150px] truncate">${escapeHtml(item.title)}</h4>
                                    <div class="flex items-center space-x-1 mt-0.5">
                                        <span class="text-[10px] text-slate-400">${item.date}</span>
                                        ${tagsHtml}
                                    </div>
                                </div>
                            </div>
                            <div class="flex items-center space-x-1">
                                <a href="${item.data}" download="${escapeHtml(item.title)}" class="p-1.5 text-indigo-600 dark:text-indigo-400 hover:bg-indigo-50 rounded-lg"><i data-lucide="download" class="w-3.5 h-3.5"></i></a>
                                <button onclick="deleteSavedAsset(${index})" class="p-1.5 text-slate-400 hover:text-red-500 rounded-lg"><i data-lucide="trash-2" class="w-3.5 h-3.5"></i></button>
                            </div>
                        </div>
                    `;
                } else {
                    const itemExt = item.ext || 'txt';
                    card.innerHTML = `
                        <div class="flex items-center justify-between">
                            <div>
                                <h4 class="font-bold text-xs max-w-[180px] truncate">${escapeHtml(item.title)}</h4>
                                <div class="flex items-center space-x-1 mt-0.5">
                                    <span class="text-[10px] text-slate-400">${item.date}</span>
                                    ${tagsHtml}
                                </div>
                            </div>
                            <div class="flex items-center space-x-1">
                                <button onclick="copyToClipboard('${escapeJs(item.content)}')" class="p-1.5 text-indigo-600 dark:text-indigo-400 hover:bg-indigo-50 rounded-lg"><i data-lucide="copy" class="w-3.5 h-3.5"></i></button>
                                <button onclick="downloadStorageTextItem('${escapeJs(item.content)}', '${escapeJs(item.title)}', '${escapeJs(itemExt)}')" class="p-1.5 text-indigo-600 dark:text-indigo-400 hover:bg-indigo-50 rounded-lg"><i data-lucide="download" class="w-3.5 h-3.5"></i></button>
                                <button onclick="deleteSavedAsset(${index})" class="p-1.5 text-slate-400 hover:text-red-500 rounded-lg"><i data-lucide="trash-2" class="w-3.5 h-3.5"></i></button>
                            </div>
                        </div>
                        <div class="bg-slate-50 dark:bg-slate-950 border border-slate-200 dark:border-slate-800 rounded-lg p-2 text-[11px] code-font max-h-20 overflow-y-auto break-all">${escapeHtml(item.content)}</div>
                    `;
                }
                container.appendChild(card);
            });
            lucide.createIcons();
        }

        function downloadStorageTextItem(content, title, ext) {
            const cleanFilename = title.endsWith(`.${ext}`) ? title : `${title}.${ext}`;
            downloadBlob(content, cleanFilename, getMimeTypeByExt(ext));
        }

        function deleteSavedAsset(index) {
            if (confirm('削除しますか？')) {
                window.savedAssets.splice(index, 1);
                localStorage.setItem(getKey('app_saved_assets'), JSON.stringify(window.savedAssets));
                if (window.saveToCloud) window.saveToCloud();
                renderSavedAssets();
            }
        }

        // --- Helpers ---
        function downloadBlob(content, filename, contentType) {
            const blob = new Blob([content], { type: contentType });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = filename;
            a.click();
            URL.revokeObjectURL(url);
        }

        function copyToClipboard(text) {
            navigator.clipboard.writeText(text).then(() => alert('コピーしました'));
        }

        function escapeHtml(str) {
            return str.replace(/[&<>"']/g, m => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' }[m]));
        }

        function escapeJs(str) {
            return str.replace(/\\/g, '\\\\').replace(/'/g, "\\'").replace(/"/g, '\\"').replace(/\n/g, '\\n');
        }
